function doGet(e) {
  try {
    const action = getRequestAction(e, 'getMenu');

    if (action === 'health') {
      return jsonOut({ ok: true, service: 'carorural-api' });
    }

    if (action === 'getMenu') {
      const menuId = (e && e.parameter && e.parameter.menu_id) ? e.parameter.menu_id : DEFAULT_MENU_ID;
      const data = getMenuRows(menuId);
      return jsonOut({ ok: true, menu_id: menuId, rows: data });
    }

    if (action === 'getLudoteca') {
      const data = getLudotecaRows();
      return jsonOut({ ok: true, rows: data });
    }

    if (action === 'getAventura' || action === 'obtenerRutas') {
      const rutas = getAventuraRows();
      return jsonOut({ ok: true, rutas: rutas, rows: rutas });
    }

    if (action === 'getPlaylists') {
      return jsonOut({ ok: true, playlists: getPlaylistsRows() });
    }

    if (action === 'getPlaylistEmbeds') {
      return jsonOut({ ok: true, playlists: getPlaylistsRows() });
    }

    if (action === 'getCanciones') {
      const idPlaylist = (e && e.parameter && e.parameter.id_playlist) ? e.parameter.id_playlist : '';
      return jsonOut({ ok: true, canciones: getCancionesRows(idPlaylist) });
    }

    if (action === 'getMuroRecuerdos') {
      return jsonOut({ ok: true, recuerdos: getMuroRecuerdosRows() });
    }

    if (action === 'getYincanas') {
      return jsonOut({ ok: true, yincanas: getYincanasRows() });
    }

    if (action === 'getNuestrasRecetas' || action === 'getRecetas') {
      return jsonOut({ ok: true, recetas: getNuestrasRecetasRows() });
    }

    if (action === 'getAmigos') {
      return jsonOut({ ok: true, miembros: getAmigosRows() });
    }

    if (action === 'getNuestroHistorico') {
      return jsonOut({ ok: true, menus: getNuestroHistoricoRows() });
    }

    if (action === 'getYincanaMedia') {
      const fileId = safeStr(e && e.parameter && e.parameter.file_id);
      if (!fileId) return jsonOut({ ok: false, error: 'Falta file_id' });
      try {
        const file = DriveApp.getFileById(fileId);
        const blob = file.getBlob();
        const base64 = Utilities.base64Encode(blob.getBytes());
        return jsonOut({ ok: true, data: base64, mimeType: blob.getContentType(), name: file.getName() });
      } catch (err) {
        return jsonOut({ ok: false, error: String(err && err.message ? err.message : err) });
      }
    }

    return jsonOut({ ok: false, error: 'Accion GET no soportada' });
  } catch (err) {
    return jsonOut({ ok: false, error: String(err && err.message ? err.message : err) });
  }
}

function doPost(e) {
  try {
    const payload = parseJsonBody(e);
    const action = safeStr(payload.action || payload.accion);

    if (action === 'registerDevice') {
      const result = registerDeviceToken(payload);
      return jsonOut({ ok: true, action: 'registerDevice', token_saved: result.token_saved });
    }

    if (action === 'uploadImage') {
      if (!payload.data || !payload.mimeType) {
        return jsonOut({ ok: false, error: 'Faltan campos: data, mimeType' });
      }

      const targetFolderId = safeStr(payload.folderId) || DRIVE_FOLDER_ID;
      if (!targetFolderId) {
        return jsonOut({ ok: false, error: 'No hay folderId configurado' });
      }

      let folder;
      try {
        folder = DriveApp.getFolderById(targetFolderId);
      } catch (err) {
        return jsonOut({ ok: false, error: 'Folder de Drive no accesible: ' + String(err && err.message ? err.message : err) });
      }

      const blob = Utilities.newBlob(
        Utilities.base64Decode(payload.data),
        payload.mimeType,
        payload.name || ('foto_' + new Date().getTime() + '.jpg')
      );

      const file = folder.createFile(blob);
      file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
      try {
        if (typeof file.setSecurityUpdateEnabled === 'function') {
          file.setSecurityUpdateEnabled(false);
        }
      } catch (e) {
      }

      return jsonOut({
        ok: true,
        action: 'uploadImage',
        url: 'https://drive.google.com/uc?export=view&id=' + file.getId(),
        file_id: file.getId(),
        file_name: file.getName(),
        folder_id: targetFolderId
      });
    }

    const lock = LockService.getScriptLock();
    lock.waitLock(30000);

    try {
      if (action === 'saveMenu') {
        const menuId = payload.menu_id || DEFAULT_MENU_ID;
        const updatedBy = payload.updated_by || 'anonimo';
        const rows = Array.isArray(payload.rows) ? payload.rows : [];

        replaceMenuRows(menuId, rows, updatedBy);
        notifyChangeSafe('saveMenu', payload);

        return jsonOut({
          ok: true,
          action: 'saveMenu',
          menu_id: menuId,
          saved_rows: rows.length
        });
      }

      if (action === 'clearMenu') {
        const menuId = payload.menu_id || DEFAULT_MENU_ID;
        clearMenuRows(menuId);
        notifyChangeSafe('clearMenu', payload);

        return jsonOut({
          ok: true,
          action: 'clearMenu',
          menu_id: menuId
        });
      }

      if (action === 'saveLudoteca') {
        const updatedBy = payload.updated_by || 'anonimo';
        const rows = Array.isArray(payload.rows) ? payload.rows : [];

        replaceLudotecaRows(rows, updatedBy);
        notifyChangeSafe('saveLudoteca', payload);

        return jsonOut({
          ok: true,
          action: 'saveLudoteca',
          saved_rows: rows.length
        });
      }

      if (action === 'guardarRuta' || action === 'saveAventuraRuta') {
        const rutaEntrada = payload.datos || payload.ruta || payload.row || payload;
        const ruta = normalizeAventuraRow(rutaEntrada);
        saveAventuraRuta(ruta);
        notifyChangeSafe('guardarRuta', payload);

        return jsonOut({
          ok: true,
          action: 'guardarRuta',
          id_ruta: ruta.id_ruta
        });
      }

      if (action === 'eliminarRuta' || action === 'deleteAventuraRuta') {
        const idRuta = safeStr(payload.Id_ruta || payload.id_ruta);
        if (!idRuta) {
          return jsonOut({ ok: false, error: 'Falta Id_ruta o id_ruta' });
        }

        const removed = deleteAventuraRuta(idRuta);
        if (removed > 0) notifyChangeSafe('eliminarRuta', payload);
        return jsonOut({
          ok: true,
          action: 'eliminarRuta',
          id_ruta: idRuta,
          removed_rows: removed
        });
      }

      if (action === 'savePlaylists') {
        const rows = Array.isArray(payload.playlists) ? payload.playlists : [];
        savePlaylistsRows(rows);
        notifyChangeSafe('savePlaylists', payload);
        return jsonOut({ ok: true, action: 'savePlaylists', saved: rows.length });
      }

      if (action === 'savePlaylistEmbed') {
        const row = payload.row || payload.playlist || payload;
        const saved = savePlaylistEmbedRow(row);
        notifyChangeSafe('savePlaylistEmbed', payload);
        return jsonOut({ ok: true, action: 'savePlaylistEmbed', id_playlist: saved.id_playlist });
      }

      if (action === 'deletePlaylistEmbed') {
        const idPlaylist = safeStr(payload.id_playlist || payload.id || payload.idPlaylist);
        if (!idPlaylist) {
          return jsonOut({ ok: false, error: 'Falta id_playlist' });
        }
        const removed = deletePlaylistEmbedRow(idPlaylist);
        if (removed > 0) notifyChangeSafe('deletePlaylistEmbed', payload);
        return jsonOut({ ok: true, action: 'deletePlaylistEmbed', id_playlist: idPlaylist, removed_rows: removed });
      }

      if (action === 'saveCanciones') {
        const idPlaylist = safeStr(payload.id_playlist);
        const rows = Array.isArray(payload.canciones) ? payload.canciones : [];
        if (!idPlaylist) {
          return jsonOut({ ok: false, error: 'Falta id_playlist' });
        }
        saveCancionesRows(idPlaylist, rows);
        notifyChangeSafe('saveCanciones', payload);
        return jsonOut({ ok: true, action: 'saveCanciones', id_playlist: idPlaylist, saved: rows.length });
      }

      if (action === 'saveMuroRecuerdo') {
        const row = payload.row || payload.recuerdo || payload;
        const saved = saveMuroRecuerdoRow(row);
        notifyChangeSafe('saveMuroRecuerdo', payload);
        return jsonOut({ ok: true, action: 'saveMuroRecuerdo', recuerdo: saved });
      }

      if (action === 'addMuroRecuerdoFotos') {
        const idRecuerdo = safeStr(payload.id_recuerdo);
        const fotos = Array.isArray(payload.fotos) ? payload.fotos : [];
        if (!idRecuerdo) {
          return jsonOut({ ok: false, error: 'Falta id_recuerdo' });
        }
        const added = addMuroRecuerdoFotos(idRecuerdo, fotos);
        if (added > 0) notifyChangeSafe('addMuroRecuerdoFotos', payload);
        return jsonOut({ ok: true, action: 'addMuroRecuerdoFotos', id_recuerdo: idRecuerdo, added: added });
      }

      if (action === 'deleteMuroRecuerdo') {
        const idRecuerdo = safeStr(payload.id_recuerdo || payload.id || payload.idRecuerdo);
        if (!idRecuerdo) {
          return jsonOut({ ok: false, error: 'Falta id_recuerdo' });
        }
        const result = deleteMuroRecuerdo(idRecuerdo);
        if (result.removedRecuerdos > 0 || result.removedFotos > 0) notifyChangeSafe('deleteMuroRecuerdo', payload);
        return jsonOut({
          ok: true,
          action: 'deleteMuroRecuerdo',
          id_recuerdo: idRecuerdo,
          removed_recuerdos: result.removedRecuerdos,
          removed_fotos: result.removedFotos
        });
      }

      if (action === 'deleteMuroFoto') {
        const idFoto = safeStr(payload.id_foto || payload.id || payload.idFoto);
        const idRecuerdo = safeStr(payload.id_recuerdo || payload.idRecuerdo);
        if (!idFoto) {
          return jsonOut({ ok: false, error: 'Falta id_foto' });
        }
        const result = deleteMuroFoto(idFoto, idRecuerdo);
        if (result.removedFotos > 0) notifyChangeSafe('deleteMuroFoto', payload);
        return jsonOut({
          ok: true,
          action: 'deleteMuroFoto',
          id_foto: idFoto,
          id_recuerdo: result.idRecuerdo,
          removed_fotos: result.removedFotos
        });
      }

      if (action === 'saveYincana') {
        const row = payload.row || payload.yincana || payload;
        const saved = saveYincanaRow(row);
        notifyChangeSafe('saveYincana', payload);
        return jsonOut({ ok: true, action: 'saveYincana', yincana: saved });
      }

      if (action === 'deleteYincana') {
        const idYincana = safeStr(payload.id_yincana || payload.id || payload.idYincana);
        if (!idYincana) {
          return jsonOut({ ok: false, error: 'Falta id_yincana' });
        }
        const removed = deleteYincanaRow(idYincana);
        if (removed > 0) notifyChangeSafe('deleteYincana', payload);
        return jsonOut({ ok: true, action: 'deleteYincana', id_yincana: idYincana, removed_rows: removed });
      }

      if (action === 'saveNuestrasReceta' || action === 'saveReceta') {
        const row = payload.row || payload.receta || payload;
        const saved = saveNuestrasRecetaRow(row);
        notifyChangeSafe('saveReceta', payload);
        return jsonOut({ ok: true, action: 'saveNuestrasReceta', receta: saved });
      }

      if (action === 'deleteNuestrasReceta' || action === 'deleteReceta') {
        const idReceta = safeStr(payload.id_receta || payload.id || payload.idReceta);
        if (!idReceta) {
          return jsonOut({ ok: false, error: 'Falta id_receta' });
        }
        const removed = deleteNuestrasRecetaRow(idReceta);
        if (removed > 0) notifyChangeSafe('deleteReceta', payload);
        return jsonOut({ ok: true, action: 'deleteNuestrasReceta', id_receta: idReceta, removed_rows: removed });
      }

      if (action === 'saveAmigo') {
        const row = payload.row || payload.miembro || payload;
        const saved = saveAmigoRow(row);
        notifyChangeSafe('saveAmigo', payload);
        return jsonOut({ ok: true, action: 'saveAmigo', miembro: saved });
      }

      if (action === 'deleteAmigo') {
        const idMiembro = safeStr(payload.id_miembro || payload.id || payload.idMiembro);
        if (!idMiembro) {
          return jsonOut({ ok: false, error: 'Falta id_miembro' });
        }
        const removed = deleteAmigoRow(idMiembro);
        if (removed > 0) notifyChangeSafe('deleteAmigo', payload);
        return jsonOut({ ok: true, action: 'deleteAmigo', id_miembro: idMiembro, removed_rows: removed });
      }

      if (action === 'saveNuestroHistoricoMenu') {
        const row = payload.row || payload.menu || payload;
        const saved = saveNuestroHistoricoMenuRow(row);
        notifyChangeSafe('saveNuestroHistoricoMenu', payload);
        return jsonOut({ ok: true, action: 'saveNuestroHistoricoMenu', menu: saved });
      }

      if (action === 'deleteNuestroHistoricoMenu') {
        const idMenu = safeStr(payload.id_menu || payload.id || payload.idMenu);
        if (!idMenu) {
          return jsonOut({ ok: false, error: 'Falta id_menu' });
        }
        const removed = deleteNuestroHistoricoMenuRow(idMenu);
        if (removed > 0) notifyChangeSafe('deleteNuestroHistoricoMenu', payload);
        return jsonOut({ ok: true, action: 'deleteNuestroHistoricoMenu', id_menu: idMenu, removed_rows: removed });
      }

      return jsonOut({ ok: false, error: 'Accion POST no soportada' });
    } finally {
      lock.releaseLock();
    }

  } catch (err) {
    return jsonOut({ ok: false, error: String(err && err.message ? err.message : err) });
  }
}

function getMenuRows(menuId) {
  const sheet = getSheetReady();
  const values = sheet.getDataRange().getValues();
  if (values.length <= 1) return [];

  const out = [];
  for (let i = 1; i < values.length; i++) {
    const r = values[i];
    const item = {
      menu_id: safeStr(r[0]),
      dia_id: safeStr(r[1]),
      fecha_iso: safeStr(r[2]),
      dia_semana: safeStr(r[3]),
      momento: safeStr(r[4]),
      item_orden: Number(r[5] || 0),
      comida: safeStr(r[6]),
      foto_url: safeStr(r[7]),
      updated_at: safeStr(r[8]),
      updated_by: safeStr(r[9]),
      deleted: toBool(r[10])
    };

    if (item.menu_id === menuId && !item.deleted) {
      out.push(item);
    }
  }

  out.sort(function(a, b) {
    if (a.fecha_iso < b.fecha_iso) return -1;
    if (a.fecha_iso > b.fecha_iso) return 1;
    if (a.momento < b.momento) return -1;
    if (a.momento > b.momento) return 1;
    return (a.item_orden || 0) - (b.item_orden || 0);
  });

  return out;
}

function replaceMenuRows(menuId, rows, updatedBy) {
  const sheet = getSheetReady();

  clearMenuRows(menuId);

  if (!rows.length) return;

  const nowIso = new Date().toISOString();
  const toAppend = rows.map(function(r) {
    return [
      menuId,
      safeStr(r.dia_id),
      safeStr(r.fecha_iso),
      safeStr(r.dia_semana),
      safeStr(r.momento),
      Number(r.item_orden || 0),
      safeStr(r.comida),
      safeStr(r.foto_url),
      nowIso,
      updatedBy,
      toBool(r.deleted) ? true : false
    ];
  });

  sheet.getRange(sheet.getLastRow() + 1, 1, toAppend.length, HEADERS.length).setValues(toAppend);
}

function clearMenuRows(menuId) {
  const sheet = getSheetReady();
  const lastRow = sheet.getLastRow();
  if (lastRow <= 1) return;

  const range = sheet.getRange(2, 1, lastRow - 1, HEADERS.length);
  const values = range.getValues();

  const keep = [];
  for (let i = 0; i < values.length; i++) {
    if (safeStr(values[i][0]) !== menuId) {
      keep.push(values[i]);
    }
  }

  sheet.clearContents();
  sheet.getRange(1, 1, 1, HEADERS.length).setValues([HEADERS]);

  if (keep.length) {
    sheet.getRange(2, 1, keep.length, HEADERS.length).setValues(keep);
  }
}

function getLudotecaRows() {
  const sheet = getLudotecaSheetReady();
  const values = sheet.getDataRange().getValues();
  if (values.length <= 1) return [];

  const out = [];
  for (let i = 1; i < values.length; i++) {
    const r = values[i];
    const item = {
      id: safeStr(r[0]),
      favorito: toBool(r[1]),
      created_at: safeStr(r[2]),
      updated_at: safeStr(r[3]),
      deleted: toBool(r[4]),
      nombre: safeStr(r[5]),
      tipo: safeStr(r[6]),
      imagen: safeStr(r[7]),
      instrucciones: safeStr(r[8]),
      jugadores: safeStr(r[9]),
      material: safeStr(r[10])
    };

    if (!item.deleted) {
      out.push(item);
    }
  }

  out.sort(function(a, b) {
    if (a.favorito !== b.favorito) return a.favorito ? -1 : 1;
    return a.nombre.localeCompare(b.nombre, 'es');
  });

  return out;
}

function replaceLudotecaRows(rows, updatedBy) {
  const sheet = getLudotecaSheetReady();

  sheet.clearContents();
  sheet.getRange(1, 1, 1, LUDOTECA_HEADERS.length).setValues([LUDOTECA_HEADERS]);

  if (!rows.length) return;

  const nowIso = new Date().toISOString();

  const toAppend = rows.map(function(r) {
    const id = safeStr(r.id) || ('j_' + new Date().getTime() + '_' + Math.floor(Math.random() * 100000));
    const createdAt = safeStr(r.created_at) || nowIso;

    return [
      id,
      toBool(r.favorito) ? true : false,
      createdAt,
      nowIso,
      toBool(r.deleted) ? true : false,
      safeStr(r.nombre),
      safeStr(r.tipo),
      safeStr(r.imagen),
      safeStr(r.instrucciones),
      safeStr(r.jugadores),
      safeStr(r.material)
    ];
  });

  sheet.getRange(2, 1, toAppend.length, LUDOTECA_HEADERS.length).setValues(toAppend);
}

function getAventuraRows() {
  const sheet = getAventuraSheetReady();
  const values = sheet.getDataRange().getValues();
  if (values.length <= 1) return [];

  const out = [];
  for (let i = 1; i < values.length; i++) {
    const r = values[i];
    const item = {
      id_ruta: safeStr(r[0]),
      fecha: safeStr(r[1]),
      nombre_ruta: safeStr(r[2]),
      duracion_seg: Number(r[3] || 0),
      distancia_km: Number(r[4] || 0),
      vel_media_kmh: Number(r[5] || 0),
      vel_max_kmh: Number(r[6] || 0),
      desnivel_m: Number(r[7] || 0),
      pasos_totales: Number(r[8] || 0),
      ruta_gps: safeStr(r[9]),
      fotos_json: safeStr(r[10])
    };

    if (item.id_ruta) {
      out.push(item);
    }
  }

  out.sort(function(a, b) {
    if (a.fecha > b.fecha) return -1;
    if (a.fecha < b.fecha) return 1;
    return String(b.id_ruta).localeCompare(String(a.id_ruta), 'es');
  });

  return out;
}

function saveAventuraRuta(ruta) {
  const sheet = getAventuraSheetReady();

  // Si ya existe ese id_ruta, primero lo quitamos para evitar duplicados.
  deleteAventuraRuta(ruta.id_ruta);

  const row = [
    safeStr(ruta.id_ruta),
    safeStr(ruta.fecha),
    safeStr(ruta.nombre_ruta),

  };

function normalizeJsonText(v) {
  if (v === null || v === undefined || v === '') return '[]';
  if (typeof v === 'string') return v;

  try {
    return JSON.stringify(v);
  } catch (err) {
    return '[]';
  }
}

}

function getMuroRecuerdosRows() {
  var recuerdosSheet = getMuroRecuerdosSheetReady();
  var fotosSheet = getMuroRecuerdosFotosSheetReady();

  var recuerdosValues = recuerdosSheet.getDataRange().getValues();
  var fotosValues = fotosSheet.getDataRange().getValues();

  var fotosByRecuerdo = {};
  for (var i = 1; i < fotosValues.length; i++) {
    var fr = fotosValues[i];
    var idFoto = safeStr(fr[0]);
    var idRecuerdo = safeStr(fr[1]);
    var fileId = safeStr(fr[2]);
    var orden = Number(fr[3] || 0);
    var esPortada = toBool(fr[4]);
    var createdAt = safeStr(fr[5]);
    var deleted = toBool(fr[6]);
    if (!idFoto || !idRecuerdo || !fileId || deleted) continue;

    if (!fotosByRecuerdo[idRecuerdo]) fotosByRecuerdo[idRecuerdo] = [];
    fotosByRecuerdo[idRecuerdo].push({
      id_foto: idFoto,
      id_recuerdo: idRecuerdo,
      file_id: fileId,
      orden: orden,
      es_portada: esPortada,
      created_at: createdAt,
      url: 'https://drive.google.com/uc?export=view&id=' + fileId
    });
  }

  Object.keys(fotosByRecuerdo).forEach(function (id) {
    fotosByRecuerdo[id].sort(function (a, b) {
      if (a.orden !== b.orden) return a.orden - b.orden;
      return String(a.id_foto).localeCompare(String(b.id_foto), 'es');
    });
  });

  var out = [];
  for (var j = 1; j < recuerdosValues.length; j++) {
    var rr = recuerdosValues[j];
    var rid = safeStr(rr[0]);
    if (!rid) continue;
    var deletedRec = toBool(rr[8]);
    if (deletedRec) continue;

    var fotos = fotosByRecuerdo[rid] || [];
    var coverFileId = safeStr(rr[4]);
    if (!coverFileId) {
      var portada = fotos.find(function (f) { return !!f.es_portada; }) || fotos[0] || null;
      coverFileId = portada ? portada.file_id : '';
    }

    out.push({
      id_recuerdo: rid,
      nombre: safeStr(rr[1]),
      fecha: safeStr(rr[2]),
      descripcion: safeStr(rr[3]),
      cover_file_id: coverFileId,
      foto_count: Number(rr[5] || fotos.length || 0),
      created_at: safeStr(rr[6]),
      updated_at: safeStr(rr[7]),
      fotos: fotos
    });
  }

  out.sort(function (a, b) {
    if (a.fecha > b.fecha) return -1;
    if (a.fecha < b.fecha) return 1;
    return String(b.id_recuerdo).localeCompare(String(a.id_recuerdo), 'es');
  });

  return out;
}

function addMuroRecuerdoFotos(idRecuerdo, fotos) {
  var fotosSheet = getMuroRecuerdosFotosSheetReady();
  if (!Array.isArray(fotos) || !fotos.length) return 0;

  var nowIso = new Date().toISOString();
  var toAppend = [];
  for (var i = 0; i < fotos.length; i++) {
    var f = fotos[i] || {};
    var idFoto = safeStr(f.id_foto) || ('RF_' + new Date().getTime() + '_' + Math.floor(Math.random() * 100000));
    var fileId = safeStr(f.file_id);
    if (!fileId) continue;
    toAppend.push([
      idFoto,
      idRecuerdo,
      fileId,
      Number(f.orden || (i + 1)),
      toBool(f.es_portada) ? true : false,
      safeStr(f.created_at) || nowIso,
      false
    ]);
  }

  if (!toAppend.length) return 0;

  fotosSheet.getRange(fotosSheet.getLastRow() + 1, 1, toAppend.length, MURO_RECUERDOS_FOTOS_HEADERS.length).setValues(toAppend);
  refreshMuroRecuerdoAggregate(idRecuerdo);
  return toAppend.length;
}

function refreshMuroRecuerdoAggregate(idRecuerdo) {
  var recuerdosSheet = getMuroRecuerdosSheetReady();
  var fotosSheet = getMuroRecuerdosFotosSheetReady();

  var recuerdosValues = recuerdosSheet.getDataRange().getValues();
  var recRowIdx = -1;
  for (var i = 1; i < recuerdosValues.length; i++) {
    if (safeStr(recuerdosValues[i][0]) === idRecuerdo) {
      recRowIdx = i + 1;
      break;
    }
  }
  if (recRowIdx < 0) return;

  var fotosValues = fotosSheet.getDataRange().getValues();
  var fotos = [];
  for (var j = 1; j < fotosValues.length; j++) {
    var row = fotosValues[j];
    if (safeStr(row[1]) !== idRecuerdo) continue;
    if (toBool(row[6])) continue;
    var fileId = safeStr(row[2]);
    if (!fileId) continue;
    fotos.push({
      file_id: fileId,
      orden: Number(row[3] || 0),
      es_portada: toBool(row[4])
    });
  }

  fotos.sort(function (a, b) {
    if (a.orden !== b.orden) return a.orden - b.orden;
    return String(a.file_id).localeCompare(String(b.file_id), 'es');
  });

  var portada = fotos.find(function (f) { return !!f.es_portada; }) || fotos[0] || null;
  var coverFileId = portada ? portada.file_id : '';
  recuerdosSheet.getRange(recRowIdx, 5).setValue(coverFileId);
  recuerdosSheet.getRange(recRowIdx, 6).setValue(fotos.length);
  recuerdosSheet.getRange(recRowIdx, 8).setValue(new Date().toISOString());
}

function deleteMuroRecuerdo(idRecuerdo) {
  var recuerdosSheet = getMuroRecuerdosSheetReady();
  var fotosSheet = getMuroRecuerdosFotosSheetReady();
  var nowIso = new Date().toISOString();

  var recuerdosValues = recuerdosSheet.getDataRange().getValues();
  var removedRecuerdos = 0;
  for (var i = 1; i < recuerdosValues.length; i++) {
    if (safeStr(recuerdosValues[i][0]) === idRecuerdo) {
      recuerdosSheet.getRange(i + 1, 9).setValue(true);
      recuerdosSheet.getRange(i + 1, 8).setValue(nowIso);
      removedRecuerdos += 1;
    }
  }

  var fotosValues = fotosSheet.getDataRange().getValues();
  var removedFotos = 0;
  for (var j = 1; j < fotosValues.length; j++) {
    if (safeStr(fotosValues[j][1]) === idRecuerdo && !toBool(fotosValues[j][6])) {
      fotosSheet.getRange(j + 1, 7).setValue(true);
      removedFotos += 1;
    }
  }

  return { removedRecuerdos: removedRecuerdos, removedFotos: removedFotos };
}

function deleteMuroFoto(idFoto, idRecuerdoHint) {
  var fotosSheet = getMuroRecuerdosFotosSheetReady();
  var values = fotosSheet.getDataRange().getValues();
  var removedFotos = 0;
  var idRecuerdo = safeStr(idRecuerdoHint);

  for (var i = 1; i < values.length; i++) {
    var row = values[i];
    if (safeStr(row[0]) !== idFoto) continue;

    if (!toBool(row[6])) {
      fotosSheet.getRange(i + 1, 7).setValue(true);
      removedFotos += 1;
    }
    break;
  }

  if (idRecuerdo) {
    refreshMuroRecuerdoAggregate(idRecuerdo);
  }

  return { idRecuerdo: idRecuerdo, removedFotos: removedFotos };
}

  try {
    var meta = buildPushMeta(action, payload);

    });

  }
}

function sendPushToRegisteredDevices(title, body, dataObj) {
  var serverKey = safeStr(PropertiesService.getScriptProperties().getProperty(SCRIPT_PROP_FCM_SERVER_KEY));
  if (!serverKey) {
    return { ok: false, reason: 'FCM_SERVER_KEY no configurada' };
  }

  var sheet = getDeviceTokensSheetReady();
  var rows = sheet.getDataRange().getValues();
  if (rows.length <= 1) {
    return { ok: true, sent: 0 };
  }

  var sent = 0;
  for (var i = 1; i < rows.length; i++) {
    var token = safeStr(rows[i][0]);
    var isActive = toBool(rows[i][6]);
    if (!token || !isActive) continue;

    var result = sendSingleFcmLegacy(serverKey, token, title, body, dataObj);
    if (result.ok) {
      sent += 1;
      sheet.getRange(i + 1, 6).setValue(new Date().toISOString());
      continue;
    }

    if (result.invalidateToken) {
      sheet.getRange(i + 1, 7).setValue(false);
      sheet.getRange(i + 1, 6).setValue(new Date().toISOString());
    }
  }

  return { ok: true, sent: sent };
}

function sendSingleFcmLegacy(serverKey, token, title, body, dataObj) {
  var payload = {
    to: token,
    priority: 'high',
    notification: {
      title: title,
      body: body,
      sound: 'default'
    },
    data: dataObj || {}
  };

  var res = UrlFetchApp.fetch(FCM_LEGACY_ENDPOINT, {
    method: 'post',
    contentType: 'application/json; charset=utf-8',
    muteHttpExceptions: true,
    headers: {
      Authorization: 'key=' + serverKey
    },
    payload: JSON.stringify(payload)
  });

  var status = Number(res.getResponseCode() || 0);
  var text = safeStr(res.getContentText());
  if (status >= 200 && status < 300) {
    try {
      var parsed = JSON.parse(text || '{}');
      var first = parsed && parsed.results && parsed.results[0] ? parsed.results[0] : null;
      var err = first && first.error ? String(first.error) : '';
      if (!err) {
        return { ok: true };
      }
      var invalid = (err === 'NotRegistered' || err === 'InvalidRegistration');
      return { ok: false, invalidateToken: invalid, error: err };
    } catch (e) {
      return { ok: true };
    }
  }

  var invalidHttp = (status === 404 || status === 410);
  return { ok: false, invalidateToken: invalidHttp, error: 'HTTP ' + status };
}
